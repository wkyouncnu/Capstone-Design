%  #1 x위치  #2 제목  #3 왼쪽부호(+1/-1)  #4 오른쪽부호  #5 왼쪽값  #6 오른쪽값
\newcommand{\boat}[6]{%
  \draw[ink, line width=0.6pt] (#1-4.2,-5) rectangle (#1-1.4,7);
  \draw[ink, line width=0.6pt] (#1+1.4,-5) rectangle (#1+4.2,7);
  \draw[ink, line width=0.5pt] (#1-4.2,4) -- (#1+4.2,4);
  % 추력 화살표 — 선체 밖(고물 아래)에만 그린다. 물을 미는 방향이다
  \draw[sig, guid, line width=1.0pt]
       (#1-2.8,{-5.5-6*(1-(#3))/2}) -- (#1-2.8,{-5.5-6*(1+(#3))/2});
  \draw[sig, ctrl, line width=1.0pt]
       (#1+2.8,{-5.5-6*(1-(#4))/2}) -- (#1+2.8,{-5.5-6*(1+(#4))/2});
  \node[font=\small\bfseries, anchor=south] at (#1,9.5) {#2};
  \node[note, anchor=north] at (#1,-15) {왼쪽 #5};
  \node[note, anchor=north] at (#1,-20) {오른쪽 #6};
}

\boat{0}  {\texttt{w} 전진}  {1}{1}  {$+200$}{$+200$}
