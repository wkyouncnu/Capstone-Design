\begin{document}
\begin{tikzpicture}[x=1mm, y=1mm, node distance=9mm]

% ---- 앞먹임 경로 -----------------------------------------------------------
\node[blk]                        (ref)  {목표 위치};
\node[gblk, right=10mm of ref]    (rm)   {기준모델};
\node[sum,  right=9mm of rm]      (s1)   {};
\node[cblk, right=9mm of s1]      (dp)   {DP 제어기};
\node[tblk, right=10mm of dp]     (al)   {추력 배분};
\node[pblk, right=10mm of al]     (sh)   {선박};

\draw[sig] (ref) -- (rm);
\draw[sig] (rm)  -- node[sname, above] {$\boldsymbol{\eta}_d$} (s1);
\draw[sig] (s1)  -- node[sname, above] {$\tilde{\boldsymbol{\eta}}$} (dp);
\draw[sig] (dp)  -- node[sname, above] {$\boldsymbol{\tau}$} (al);
\draw[sig] (al)  -- node[sname, above] {$\mathbf{f}$} (sh);

\node[note, below=1.2mm of s1] {$-$};

% ---- 외란 -----------------------------------------------------------------
\node[blk, draw=warn, above=11mm of sh] (dist) {외란};
\node[note, right=1.5mm of dist] {바람 · 파랑};
\draw[sig, warn] (dist) -- (sh);

% ---- 되먹임 경로 -----------------------------------------------------------
\node[nblk, below=13mm of al]  (nav)  {항법};
\node[nblk, left=12mm of nav]  (filt) {파랑 필터};

\coordinate (shb) at ($(sh.south)+(0,-13mm)$);
\draw[wire] (sh.south) -- (shb);
\draw[sig]  (shb) -| (nav.east);
\draw[sig]  (nav) -- node[sname, below] {$\boldsymbol{\eta},\ \boldsymbol{\nu}$} (filt);
\draw[sig]  (filt.west) -| (s1.south);

% ---- 설명 -----------------------------------------------------------------
\node[note, below=1.5mm of rm.south, text width=26mm, align=center]
     {계단 목표를\\완만하게};
\node[note, below=1.5mm of filt.south, text width=28mm, align=center]
     {파도 성분만 걸러낸다};
