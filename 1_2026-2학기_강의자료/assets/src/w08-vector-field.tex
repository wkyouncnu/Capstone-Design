% 요점: 배가 어디에 있든 그 자리에서 가야 할 방향이 정해진다.
%       필드 화살표는 실제로 식 (13) 을 계산해서 그린다.
%
%   chi_d = theta + atan2(r*thetadot, rdot)
%   여기서는 rdot ~ -(r - r_d), r*thetadot ~ p_c * r_d  로 두고 방향만 그린다.
%
% 좌표: 화면 위 = 북(N), 오른쪽 = 동(E). TikZ 각은 동쪽에서 반시계,
%       theta 는 북에서 시계로 잰다 -> TikZ 각 = 90 - theta.
%       p_c > 0 은 시계방향 회전 = 화면에서도 시계방향.
%
% 고친 것 (2026-09-18) — 화살표가 r_d 와 설명 글자를 덮었고, N·E 축과 theta 호가 없어
% 본문의 "theta = atan2(E, N), 북에서 시계" 를 그림에서 확인할 수 없었다.

\documentclass[border=6pt]{standalone}
\input{capstone-style}

\begin{document}
\begin{tikzpicture}[x=1mm, y=1mm]

\def\R{34}       % 목표 반경 [mm]
\def\PC{1.1}     % 접선 성분 비중 (그림용)

% ---- N · E 축 (왼쪽 위) ----------------------------------------------------
\begin{scope}[shift={(-60,40)}]
  \draw[axis, -{Stealth[length=3.5pt,width=2.8pt]}] (0,0) -- (0,10);
  \draw[axis, -{Stealth[length=3.5pt,width=2.8pt]}] (0,0) -- (10,0);
  \node[font=\small, anchor=south] at (0,10.3) {$x$ (북)};
  \node[font=\small, anchor=west]  at (10.3,0) {$y$ (동)};
\end{scope}

% ---- 목표 원 (화살표보다 먼저 — 원 위 화살표가 가려지지 않게) ----
\draw[pathc, line width=2.2pt, opacity=0.18] (0,0) circle (\R mm);   % 옅은 띠 — 원 위 접선 화살표가 보이게

% ---- 벡터필드 -------------------------------------------------------------
% 여러 반경·각도에서 방향을 계산해 짧은 화살표로 그린다
\foreach \ang in {0,20,...,340}{
  \foreach \rr in {20,34,46}{
    \pgfmathsetmacro{\rd}{-(\rr-\R)/14}      % 반경 성분 : 목표 반경으로 끌어당긴다
    \pgfmathsetmacro{\td}{-\PC}              % 접선 성분 : 음수 = 시계 방향 (p_c > 0)
    \pgfmathsetmacro{\dir}{\ang + atan2(\td,\rd)}
    \draw[flow, line width=0.45pt, -{Stealth[length=2.6pt,width=2.0pt]}]
         (\ang:\rr mm) -- ++(\dir:4.2mm);
  }
}

% ---- 목표 원 · 중심 · r_d -------------------------------------------------
\fill[ink] (0,0) circle (1.2);
\node[note, fill=white, inner sep=0.6pt, anchor=north] at (0,-1.8) {중심};
\draw[pathc, line width=0.7pt] (0,0) -- (215:\R mm);
\node[note, pathc, fill=white, inner sep=0.8pt] at (215:25mm) {$r_d$};

% ---- theta — 북에서 시계로, 오른쪽 배까지 -------------------------------
\draw[axis, densely dotted] (0,0) -- (90:28mm);
\draw[soft, densely dotted] (0,0) -- (0:53mm);
\draw[errc, line width=0.6pt, -{Stealth[length=3pt,width=2.4pt]}] (90:11mm) arc (90:2:11mm);
\node[note, errc, fill=white, inner sep=0.8pt] at (45:14.5mm) {$\theta$};

% ---- 세 위치의 배 ---------------------------------------------------------
\fill[errc] (0:56mm) circle (1.5);
\node[note, errc, anchor=west, fill=white, inner sep=0.8pt] at (0:59mm) {$r > r_d$ — 안쪽으로};

\fill[errc] (150:14mm) circle (1.5);
\node[note, errc, anchor=south, fill=white, inner sep=0.8pt] at (150:16.5mm) {$r < r_d$ — 바깥으로};

\fill[pathc] (255:\R mm) circle (1.5);
\node[note, pathc, anchor=north, fill=white, inner sep=0.8pt] at (255:37.5mm) {$r = r_d$ — 순수 접선};

% ---- 식 -------------------------------------------------------------------
\node[font=\small, anchor=west] at (-60,-58)
     {$\chi_d = \theta + \operatorname{atan2}\!\left(r\dot{\theta},\ \dot{r}\right)$};

\node[note, anchor=west, text width=62mm] at (-6,-58)
     {$\dot{r}$ — 반경 방향 (다가갈지 멀어질지)\\
      $r\dot{\theta}$ — 접선 방향 (돌아가는 힘)\\
      두 성분의 비율이 곧 접근 각도가 된다};

\node[wnote, anchor=west, text width=124mm] at (-60,-70)
     {$p_c$ 의 \textbf{부호}가 회전 방향을 정한다 (양수 = 시계).
      \textbf{크기}가 작을수록 원에 가파르게 붙고(1-6절), 정상상태 반경 오차 $\approx p_c v \tau$ 도 작아진다(1-8절).
      실측 — 목표 25\,m 에서 $p_c$ 0.15 / 0.60 / 1.00 $\rightarrow$ 26.03 / 29.09 / 31.81\,m};

\end{tikzpicture}
\end{document}
